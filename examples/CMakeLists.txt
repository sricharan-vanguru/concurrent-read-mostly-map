# Domain policy stays outside the installed string-map library.
add_library(example_routing STATIC routing_table.cpp)
target_include_directories(example_routing PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(example_routing PUBLIC read_mostly::map)
read_mostly_warnings(example_routing)
if(READ_MOSTLY_BUILD_EXAMPLES)
    foreach(example config_reload routing)
        add_executable(${example}_example ${example}.cpp)
        target_link_libraries(${example}_example PRIVATE example_routing)
        read_mostly_warnings(${example}_example)
        if(READ_MOSTLY_BUILD_TESTS)
            add_test(NAME ${example}_example COMMAND ${example}_example)
        endif()
    endforeach()
endif()
if(READ_MOSTLY_BUILD_TESTS)
    add_executable(integration_tests ${PROJECT_SOURCE_DIR}/tests/integration_tests.cpp)
    target_link_libraries(integration_tests PRIVATE example_routing)
    read_mostly_warnings(integration_tests)
    add_test(NAME integration_tests COMMAND integration_tests)
    set_tests_properties(integration_tests PROPERTIES TIMEOUT 60)
endif()
